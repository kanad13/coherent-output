"""Search local notes without sending their contents to another service."""
import sqlite3

def search(connection, term):
    """Return titles containing the caller's search text."""
    rows = connection.execute("SELECT title FROM notes ORDER BY title").fetchall()
    return [row[0] for row in rows if term in row[0]]

def example_database():
    connection = sqlite3.connect(":memory:")
    connection.execute("CREATE TABLE notes (title TEXT NOT NULL)")
    connection.executemany("INSERT INTO notes VALUES (?)", [("Field notes",), ("Trip planning",)])
    return connection
